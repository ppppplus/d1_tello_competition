cmake_minimum_required(VERSION 3.16)
project(d1_tello_competition LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(ignition-gazebo6 REQUIRED COMPONENTS gui)
find_package(ignition-transport11 REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(nlohmann_json REQUIRED)
foreach(dep rclcpp geometry_msgs nav_msgs sensor_msgs rosgraph_msgs std_msgs std_srvs)
  find_package(${dep} REQUIRED)
endforeach()
add_library(D1CompetitionPolicy SHARED src/d1_policy.cc)
target_link_libraries(D1CompetitionPolicy ignition-gazebo6::ignition-gazebo6 Eigen3::Eigen nlohmann_json::nlohmann_json)
ament_target_dependencies(D1CompetitionPolicy rclcpp geometry_msgs nav_msgs sensor_msgs rosgraph_msgs)
add_library(D1Mission SHARED src/mission.cc)
target_link_libraries(D1Mission ignition-gazebo6::ignition-gazebo6 nlohmann_json::nlohmann_json)
ament_target_dependencies(D1Mission rclcpp geometry_msgs nav_msgs std_msgs std_srvs)
add_executable(camera_bridge src/camera_bridge.cc)
target_link_libraries(camera_bridge ignition-transport11::core)
ament_target_dependencies(camera_bridge rclcpp sensor_msgs)
install(TARGETS camera_bridge RUNTIME DESTINATION lib/${PROJECT_NAME})
add_executable(competition_gui src/gui.cc)
target_link_libraries(competition_gui ignition-gazebo6::gui)
install(TARGETS competition_gui RUNTIME DESTINATION lib/${PROJECT_NAME})
install(TARGETS D1CompetitionPolicy D1Mission LIBRARY DESTINATION lib)
install(DIRECTORY launch config worlds assets policy licenses docs DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/simulator.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/scenario.py scripts/entry_scene.py DESTINATION lib/${PROJECT_NAME})
install(FILES README.md THIRD_PARTY.md LICENSE DESTINATION share/${PROJECT_NAME})
ament_package()
